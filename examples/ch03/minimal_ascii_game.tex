\begin{lstlisting}[style=asm,caption={A complete, minimal ASCII game.}]
.option norelax
.section .data
hello_score:  .word 0

.section .rodata
title:  .asciz "HELLO PRG32\n"

.section .text
.global hello_init
.global hello_update
.global hello_draw

hello_init:
    addi sp, sp, -16
    sw   ra, 12(sp)
    call prg32_console_clear
    la   a0, title
    call prg32_console_write
    li   a0, 440            # 440 Hz: a short, pleasant beep on start
    li   a1, 60             # 60 ms duration
    call prg32_audio_beep   # silent on QEMU, audible on the board
    lw   ra, 12(sp)
    addi sp, sp, 16
    ret

hello_update:
    la   t0, hello_score
    lw   t1, 0(t0)
    addi t1, t1, 1          # count one tick per frame
    sw   t1, 0(t0)
    ret

hello_draw:
    ret
\end{lstlisting}
